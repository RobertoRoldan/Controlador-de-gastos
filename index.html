Configurar Tablas en Supabase

Copia este código y pégalo en el Editor SQL de tu panel de Supabase para activar las tablas multiusuario (categories, category_members, expenses, user_config), permisos RLS y tiempo real:

-- 1. TABLA: categories create table if not exists public.categories ( id uuid primary key default gen_random_uuid(), owner_id uuid not null references auth.users(id) on delete cascade, owner_email text not null default '', name text not null, created_at timestamptz not null default now(), updated_at timestamptz not null default now(), constraint categories_owner_name_key unique (owner_id, name) ); -- 2. TABLA INTERMEDIA: category_members (Permisos / Relación multiusuario) create table if not exists public.category_members ( id uuid primary key default gen_random_uuid(), category_id uuid not null references public.categories(id) on delete cascade, user_id uuid references auth.users(id) on delete cascade, user_email text not null, role text not null default 'editor' check (role in ('owner', 'editor', 'viewer')), created_at timestamptz not null default now(), constraint category_members_cat_email_key unique (category_id, user_email) ); -- 3. TABLA: expenses (Movimientos de cada categoría) create table if not exists public.expenses ( id uuid primary key default gen_random_uuid(), category_id uuid not null references public.categories(id) on delete cascade, "desc" text not null, amount numeric not null, date text not null, tags text not null default '', created_by uuid references auth.users(id) on delete set null, created_by_email text not null default '', created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); -- 4. TABLA: user_config (Preferencias personales) create table if not exists public.user_config ( user_id uuid primary key references auth.users(id) on delete cascade, config jsonb not null default '{}'::jsonb, updated_at timestamptz not null default now() ); -- Índices create index if not exists idx_categories_owner on public.categories(owner_id); create index if not exists idx_category_members_cat on public.category_members(category_id); create index if not exists idx_category_members_email on public.category_members(lower(user_email)); create index if not exists idx_category_members_uid on public.category_members(user_id); create index if not exists idx_expenses_category on public.expenses(category_id); create index if not exists idx_expenses_date on public.expenses(date); -- Trigger auto user_id create or replace function public.link_category_member_user_id() returns trigger language plpgsql security definer as $$ begin if new.user_id is null and new.user_email is not null then select id into new.user_id from auth.users where lower(email) = lower(trim(new.user_email)) limit 1; end if; return new; end; $$; drop trigger if exists trg_link_category_member on public.category_members; create trigger trg_link_category_member before insert or update on public.category_members for each row execute function public.link_category_member_user_id(); -- Funciones de Seguridad RLS create or replace function public.is_category_owner(p_cat_id uuid) returns boolean language sql security definer set search_path = public stable as $$ select exists (select 1 from public.categories where id = p_cat_id and owner_id = auth.uid()); $$; create or replace function public.is_category_member(p_cat_id uuid) returns boolean language sql security definer set search_path = public stable as $$ select exists (select 1 from public.category_members where category_id = p_cat_id and (user_id = auth.uid() or lower(user_email) = lower(coalesce(auth.jwt() ->> 'email', '')))); $$; create or replace function public.can_edit_category(p_cat_id uuid) returns boolean language sql security definer set search_path = public stable as $$ select exists (select 1 from public.categories where id = p_cat_id and owner_id = auth.uid()) or exists (select 1 from public.category_members where category_id = p_cat_id and (user_id = auth.uid() or lower(user_email) = lower(coalesce(auth.jwt() ->> 'email', ''))) and role in ('owner', 'editor')); $$; -- RLS alter table public.categories enable row level security; alter table public.category_members enable row level security; alter table public.expenses enable row level security; alter table public.user_config enable row level security; create policy "categories_select_policy" on public.categories for select using (owner_id = auth.uid() or public.is_category_member(id)); create policy "categories_insert_policy" on public.categories for insert with check (owner_id = auth.uid()); create policy "categories_update_policy" on public.categories for update using (owner_id = auth.uid()); create policy "categories_delete_policy" on public.categories for delete using (owner_id = auth.uid()); create policy "members_select_policy" on public.category_members for select using (public.is_category_owner(category_id) or user_id = auth.uid() or lower(user_email) = lower(coalesce(auth.jwt() ->> 'email', ''))); create policy "members_insert_policy" on public.category_members for insert with check (public.is_category_owner(category_id)); create policy "members_update_policy" on public.category_members for update using (public.is_category_owner(category_id)); create policy "members_delete_policy" on public.category_members for delete using (public.is_category_owner(category_id) or user_id = auth.uid() or lower(user_email) = lower(coalesce(auth.jwt() ->> 'email', ''))); create policy "expenses_select_policy" on public.expenses for select using (public.is_category_owner(category_id) or public.is_category_member(category_id)); create policy "expenses_insert_policy" on public.expenses for insert with check (public.can_edit_category(category_id)); create policy "expenses_update_policy" on public.expenses for update using (public.can_edit_category(category_id)); create policy "expenses_delete_policy" on public.expenses for delete using (public.can_edit_category(category_id)); create policy "user_config_all_policy" on public.user_config for all using (user_id = auth.uid()) with check (user_id = auth.uid()); -- Realtime do $$ begin if not exists (select 1 from pg_publication_tables where pubname = 'supabase_realtime' and tablename = 'categories') then alter publication supabase_realtime add table public.categories; end if; if not exists (select 1 from pg_publication_tables where pubname = 'supabase_realtime' and tablename = 'category_members') then alter publication supabase_realtime add table public.category_members; end if; if not exists (select 1 from pg_publication_tables where pubname = 'supabase_realtime' and tablename = 'expenses') then alter publication supabase_realtime add table public.expenses; end if; end $$;
