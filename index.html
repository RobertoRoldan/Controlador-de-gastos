Crear Tabla en Supabase

Copia este código y pégalo en el Editor SQL de tu panel de Supabase para activar la tabla user_finanzas:

create table public.user_finanzas ( user_id uuid primary key references auth.users(id) on delete cascade, data jsonb not null default '{}'::jsonb, config jsonb not null default '{}'::jsonb, updated_at timestamptz default now() ); alter table public.user_finanzas enable row level security; create policy "Usuarios leen sus datos" on public.user_finanzas for select using (auth.uid() = user_id); create policy "Usuarios insertan sus datos" on public.user_finanzas for insert with check (auth.uid() = user_id); create policy "Usuarios actualizan sus datos" on public.user_finanzas for update using (auth.uid() = user_id);
